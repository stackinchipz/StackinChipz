\currentdate{\today}
\lettercontent{Dear [Hiring Manager Name, or fall back to: Hiring Manager],}

\lettercontent{[Opening: name the role (and requisition ID if listed), lead with a specific, verified hook about the company or the problem the role solves, and connect your strongest qualification to it in one sentence. 2 to 3 sentences.]}

\lettercontent{[Why this company: one or two verified specifics, such as a product, a stated priority, or a recent launch, and why they matter to you. Never generic. 2 to 3 sentences.]}

\lettercontent{[Task-solving lead-in: one sentence on how you would tackle the role's main challenges, followed by evidence:]}

{\raggedright\fontspec[Path = OpenFonts/fonts/raleway/, BoldFont = Raleway-Bold]{Raleway-Medium}\fontsize{11pt}{13pt}\selectfont
\begin{itemize}
    \item \textbf{[Requirement 1]:} [concrete, quantified achievement that proves it]
    \item \textbf{[Requirement 2]:} [skill or project mapped to a must-have in the posting]
    \item \textbf{[Requirement 3]:} [evidence for a nice-to-have or initiative you took]
\end{itemize}\par}
\vspace{6pt}

\lettercontent{[Personal fit (optional): a behavioral strength from 02-behavioral-profile.md tied to how this team works. 1 to 2 sentences.]}

\lettercontent{[Close: confident and forward-looking, for example: I would welcome the chance to discuss how I could help your team reach (specific goal).]}

\needspace{5\baselineskip}
% US block format: closing and signature are left-aligned.
\closing{Sincerely,}
\signature{[YOUR NAME]}
